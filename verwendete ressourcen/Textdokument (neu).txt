            document.getElementById('search-input').addEventListener('keypress', function(e) {
                if (e.key === 'Enter') {
                    const searchTerm = this.value.trim();
                    if (searchTerm) {
                        const location = predefinedLocations[searchTerm];
                        if (location) {
                            map.setView([location.lat, location.lng], location.zoom || 14);
                            document.getElementById('sidebar').classList.add('-translate-x-full');
                        } else {
                            // Try to find a match in our locations
                            const foundLocation = locations.find(loc => 
                                loc.name.toLowerCase().includes(searchTerm.toLowerCase())
                            );
                            if (foundLocation) {
                                map.setView([foundLocation.lat, foundLocation.lng], 16);
                                document.getElementById('sidebar').classList.add('-translate-x-full');
                            } else {
                                alert('Location not found. Try "Dortmund", "Bochum", "Hochschule Bochum", or "Ruhr Universität Bochum"');
                            }
                        }
                    }
                }
            });

            // Update reports list in sidebar
            function updateReportsList() {
                const reportsList = document.getElementById('reports-list');
                reportsList.innerHTML = '';
                
                // Sort by number of reports (descending)
                const sortedLocations = [...locations].sort((a, b) => b.reports - a.reports);
                
                sortedLocations.forEach(location => {
                    let bgClass, iconClass, textClass, icon;
                    
                    if (location.type === "accessible") {
                        bgClass = 'bg-blue-50';
                        iconClass = 'bg-green-500';
                        textClass = 'text-blue-800';
                        icon = '<i class="fas fa-check"></i>';
                    } else if (location.type === "unknown") {
                        bgClass = 'bg-yellow-50';
                        iconClass = 'bg-yellow-500';
                        textClass = 'text-yellow-800';
                        icon = '<i class="fas fa-question"></i>';
                    } else {
                        bgClass = 'bg-red-50';
                        iconClass = 'bg-red-500';
                        textClass = 'text-red-800';
                        icon = '<i class="fas fa-exclamation"></i>';
                    }
                    
                    const reportItem = document.createElement('div');
                    reportItem.className = `${bgClass} p-3 rounded-lg`;
                    reportItem.innerHTML = `
                        <div class="flex justify-between items-start">
                            <div>
                                <h3 class="font-medium ${textClass}">${location.name}</h3>
                                <p class="text-sm text-gray-600">${location.reports} ${location.reports === 1 ? 'report' : 'reports'}</p>
                            </div>
                            <div class="access-icon ${iconClass}">
                                ${icon}
                            </div>
                        </div>
                        <div class="mt-2 text-sm">
                            <p>${location.description}</p>
                            <button class="mt-2 ${textClass} text-sm font-medium">View Details</button>
                        </div>
                    `;
                    
                    reportItem.addEventListener('click', () => {
                        map.setView([location.lat, location.lng], 16);
                        document.getElementById('sidebar').classList.add('-translate-x-full');
                    });
                    
                    reportsList.appendChild(reportItem);
                });
            }
            
            // Filter markers based on selected filters
            function filterMarkers() {
                const wheelchairChecked = document.getElementById('filter-wheelchair').checked;
                const noStepsChecked = document.getElementById('filter-no-steps').checked;
                const autoDoorsChecked = document.getElementById('filter-auto-doors').checked;
                const elevatorsChecked = document.getElementById('filter-elevators').checked;
                
                markers.forEach(markerObj => {
                    const location = markerObj.data;
                    let show = false;
                    
                    // If no filters are selected, show all
                    if (!wheelchairChecked && !noStepsChecked && !autoDoorsChecked && !elevatorsChecked) {
                        show = true;
                    } else {
                        // Check if location matches any selected filters
                        if (wheelchairChecked && location.features.includes("wheelchair")) {
                            show = true;
                        }
                        if (noStepsChecked && location.features.includes("no-steps")) {
                            show = true;
                        }
                        if (autoDoorsChecked && location.features.includes("auto-doors")) {
                            show = true;
                        }
                        if (elevatorsChecked && location.features.includes("elevators")) {
                            show = true;
                        }
                    }
                    
                    if (show) {
                        if (!map.hasLayer(markerObj.marker)) {
                            markerObj.marker.addTo(map);
                        }
                    } else {
                        if (map.hasLayer(markerObj.marker)) {
                            map.removeLayer(markerObj.marker);
                        }
                    }
                });
            }
