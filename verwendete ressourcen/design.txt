    <script>
        // Toggle sidebar on mobile
        document.getElementById('menuButton').addEventListener('click', function() {
            document.querySelector('.sidebar').classList.toggle('active');
        });

        // Simple tooltip for pins
        document.querySelectorAll('.pin').forEach(pin => {
            const title = pin.getAttribute('title');
            if (title) {
                const tooltip = document.createElement('div');
                tooltip.className = 'absolute bg-black text-white text-xs px-2 py-1 rounded whitespace-nowrap hidden';
                tooltip.style.top = '-30px';
                tooltip.style.left = '50%';
                tooltip.style.transform = 'translateX(-50%)';
                tooltip.textContent = title;
                pin.appendChild(tooltip);
                
                pin.addEventListener('mouseenter', () => tooltip.classList.remove('hidden'));
                pin.addEventListener('mouseleave', () => tooltip.classList.add('hidden'));
            }
        });
    </script>
